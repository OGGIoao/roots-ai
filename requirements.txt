# 创建方式 (必须严格按顺序):
#
#   conda create -n roots-car python=3.10 pip -y
#   conda activate roots-car
#   pip install torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements.txt
#   pip install -e . --no-deps        # 仓库自带的共享包 common/ (见 pyproject.toml)
#
# ⚠️ 铁律: 同一个环境里**不要混用 conda 和 pip 装 torch 系包**。
#    典型事故: conda 装的 CUDA 版被 pip 装的 CPU 版覆盖, conda 再把 CUDA 版清掉,
#    最后只剩 CPU 版 + 一个 ABI 不匹配的 torchaudio。
#
# ⚠️ torch / torchvision / torchaudio 三件套必须用**同一条命令**一起装,
#    并且都来自同一个索引, 否则版本不配套会 import 失败。
#    cu128 是 RTX 50 系 (sm_120) 的最低要求; 老显卡可换成其它 CUDA 索引,
#    没有 GPU 时去掉 --index-url 装 CPU 版即可 (大部分项目在 CPU 上也能跑)。
#
# 验证: python verify_env.py --full     (应为全绿)
# ============================================================================
