# 创建方式 (必须严格按顺序, 否则会重演旧的 ml 环境事故):
#
#   conda create -n mlc python=3.10 pip -y
#   conda activate mlc
#   pip install torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements-mlc.txt
#
# ⚠️ 铁律: 同一个环境里**不要混用 conda 和 pip 装 torch 系包**。
#    旧的 ml 环境就是因为 conda 装了 cu118 版、pip 又装了 cpu 版覆盖,
#    最后 conda 把 CUDA 版清掉, 只剩 CPU 版 + 一个 ABI 坏掉的 torchaudio。
#
# ⚠️ torch / torchvision / torchaudio 三件套必须用**同一条命令**一起装,
#    并且都来自 cu128 索引, 否则版本不配套会 import 失败。
#
# 验证: python verify_env.py --full     (应为全绿)
